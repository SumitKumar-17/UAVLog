\begin{frame}[shrink=5]{Motivation and problem statement}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Why flight logs are hard}
\begin{itemize}
  \item High-rate time series: 300 PX4 flights hold \textbf{70.7 M} samples
  \item Free text: warnings, failure flags, pilot notes (3,401 events)
  \item Vendor dialects: PX4 and ArduPilot name the same quantity differently
  \item Tools plot \textbf{one log at a time}; people ask \emph{``which flights lost a motor?''}
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Problem statement and research questions}
\begin{itemize}
  \item \textbf{RQ1} How should high-rate telemetry be stored, and at what cost?
  \item \textbf{RQ2} Which vector index and which filtering strategy retrieve events best?
  \item \textbf{RQ3} Can a model write safe, correct SQL, and what does a semantic layer add?
  \item \textbf{RQ4} Does the pipeline survive failure and scale to hundreds of thousands of logs?
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% 3 -------------------------------------------------------------
\begin{frame}[shrink=5]{Background: the database building blocks}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{TimescaleDB (time series)}
\begin{itemize}
  \item \textbf{Hypertable}: one logical table split into time \emph{chunks}
  \item \textbf{Columnstore}: old chunks stored column-wise and compressed, grouped by \code{segmentby}
  \item \textbf{Continuous aggregate}: incrementally maintained rollup (1-second min/max/mean)
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{pgvector and pgvectorscale (similarity)}
\begin{itemize}
  \item \textbf{HNSW}: layered proximity graph, fast approximate search
  \item \textbf{IVFFlat}: cluster lists, probe the nearest few
  \item \textbf{DiskANN}: compressed graph on disk (pgvectorscale)
  \item Known pitfall: \textbf{filters apply after the index scan}, so filtered queries can under-return
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{columns}[T]
\begin{column}{0.4\textwidth}
\begin{block}{Gap}
\begin{itemize}
  \item Log viewers: one flight, no semantics, no text query
  \item Time-series DBs: no ontology or vendor mapping
  \item NL-to-SQL demos: rarely guarded or measured

\begin{column}{0.48\textwidth}
\begin{block}{Design decisions}
\begin{itemize}
  \item Narrow long format: a new signal adds rows, never columns
  \item \code{telemetry} hypertable, 7-day chunks, 70.7 M rows
  \item \code{vendor\_field} and \code{field\_mapping} carry scale, offset and a note
  \item \code{event} holds text, a \code{vector(1536)} embedding and a generated \code{tsvector}
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[shrink=5]{Core contribution 2: the semantic layer}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Ontology}
\begin{itemize}
  \item \textbf{63 concepts, 55 relations}: 34 signals, 7 subsystems, 7 failure modes, 15 event kinds
  \item Signal $\to$ subsystem $\to$ failure mode $\to$ event, walked with recursive CTEs
  \item Queries say \code{battery.voltage}, not the vendor field name
\end{itemize}
\end{block}
\begin{block}{ArduPilot: a wrong mapping is worse than a gap}
\begin{itemize}
  \item Mapped: attitude, GPS, position, velocity, battery
  \item Not mapped: vibration, motors, EKF, barometric altitude, mode
\end{itemize}
\end{block}
\end{column}

\begin{frame}[shrink=5]{Method 1: pipelines that survive failure}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\centering
\fig{width=\textwidth,height=0.7\textheight,keepaspectratio}{diagram-queue.png}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Job tables in Postgres}
\begin{itemize}
  \item Status, attempts, last error, lease time per item
  \item Claimed with \code{FOR UPDATE SKIP LOCKED}: workers run concurrently
  \item Idempotent, atomic per item, failure-isolated, stale leases reclaimed
  \item Grow by enqueueing items, never by changing code
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% 8 -------------------------------------------------------------
\begin{frame}[shrink=5]{Method 2: hybrid search that routes itself}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{block}{Hybrid retrieval}
\begin{itemize}
  \item Vector (HNSW) + full text (GIN) + SQL filters
  \item Fused by \textbf{reciprocal rank fusion}, weights (1.0, 0.1)
  \item With a 10\% filter the planner chose HNSW post-filtering: \textbf{4 of 10 rows}, recall 0.37
\end{itemize}
\end{block}
\begin{block}{The router}
\begin{itemize}
  \item Estimate filtered rows with \code{EXPLAIN} on the filter alone
  \item $\le$ 1,500 rows: filter first, sort exactly; else HNSW with iterative scan
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.47\textwidth}
\centering

\end{column}

\begin{block}{Four independent defences}
\begin{itemize}
  \item \textbf{Guard} (\code{sqlglot}): one SELECT, allow-lists, SQL rebuilt from the parse tree
  \item \textbf{Role} that reads only documented views
  \item \textbf{Read-only} transaction, 8 s timeout
  \item \textbf{Cost admission}: planner estimate under 100,000
\end{itemize}

\begin{frame}[shrink=5]{Optimization: columnstore physical design}
\begin{columns}[T]
\begin{column}{0.6\textwidth}
\centering
{\scriptsize 100 flights, 26.9 M rows, median of 7 warm runs; Q2 all flights, Q3 one flight, Q4 90 days}\par
\begin{tabular}{@{}lrrrr@{}}
\toprule
variant & size & Q2 & Q3 & Q4\\
\midrule
rowstore + B-tree & 3,631 MB & 1,519 ms & 65 ms & 246 ms\\
\textbf{segment (flight, signal)} & \textbf{147 MB} & \textbf{16 ms} & 10.8 ms & \textbf{2.7 ms}\\
no segmentby & 157 MB & 29.7 ms & 13.7 ms & 3.6 ms\\
1-day chunks & 150 MB & 15.3 ms & 12.3 ms & 3.4 ms\\
30-day chunks & 143 MB & 8.3 ms & 5.7 ms & 2.4 ms\\
\bottomrule
\end{tabular}\par\vspace{0.4em}
\fig{width=0.8\textwidth,height=0.25\textheight,keepaspectratio}{code-columnstore-ddl.png}
\end{column}
\begin{column}{0.37\textwidth}
\begin{block}{Findings}
\begin{itemize}
  \item \textbf{25$\times$ smaller, up to 95$\times$ faster} cross-flight queries
  \item Design choices change size by under 10\%
  \item Chunk count and \code{segmentby} change cross-flight speed about 2$\times$
  \item Partial chunks must be recompressed: found by testing
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% 16 ------------------------------------------------------------
\begin{frame}[shrink=5]{Conclusions, limits and next steps}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Conclusions}
\begin{itemize}
  \item One PostgreSQL holds time series, ontology, vectors and guarded NL-to-SQL
  \item Physical design cut storage \textbf{32$\times$}; filtering strategy mattered more than index choice
  \item Documentation drives NL-to-SQL accuracy; four layers keep it safe
  \item Hybrid fusion did \emph{not} help: reported as found
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Limits and next steps}
\begin{itemize}
  \item 300 flights, one machine, one model, author-written questions
  \item Index-benchmark vectors are perturbed copies of real ones
  \item Next: the large run (10 k+ logs), ArduPilot at scale, concurrency benchmark
  \item Snapshot tool is untested end to end
\end{itemize}
\end{block}
\end{column}
