g AS (SELECT *, row_number() OVER (ORDER BY ts) - row_number() OVER (PARTITION BY v ORDER BY ts) AS grp FROM s)
SELECT v, min(ts) AS started, max(ts) AS ended FROM g GROUP BY v, grp ORDER BY started
